Example 02 — The Same Triangle, as a Vertex Buffer Object
Concept: The 5-step VBO pattern: glGenBuffers -> glBindBuffer -> glBufferData -> glVertexPointer(with a
byte OFFSET) -> glDrawArrays -> cleanup.

Expected output: The identical green triangle from Example 01, now stored in GPU (server) memory and
rendered from a VBO.

  #define GL_SILENCE_DEPRECATION
  #ifdef __APPLE__
  #include <GLUT/glut.h>
  #else
  #include <GL/glew.h>
  #include <GL/glut.h>
  #endif
  #include <iostream>
  #include <cmath>
  #include <cstdlib>
  using namespace std;

  /*
   * Example 02 - The Same Triangle, as a Vertex Buffer Object (VBO)
   * ------------------------------------------------------------
   * Concept: The 5-step VBO pattern:
   *   1. Initialize the vertex data in an ordinary array.
   *   2. Generate a buffer ID and bind it: glGenBuffers, glBindBuffer,
   *      glBufferData - this UPLOADS the array to the GPU's own memory.
   *   3. glEnableClientState(GL_VERTEX_ARRAY).
   *   4. glVertexPointer(...) + glDrawArrays(...) - note the pointer
   *      argument is now an OFFSET (0), not a real pointer, because the
   *      data already lives in the currently BOUND buffer.
   *   5. Clean up: disable state, unbind, and delete the buffer.
   */

  GLuint VBOid;



                                   Module 5 — Student Laboratory Manual | Page 4
void triangle() {
    // 1. initialize the triangle using an array
    GLfloat trianglevertices[] = {
        0.0f, .75f, 0.0f,
        -0.75f, 0.0f, 0.0f,
        0.75f, 0.0f, 0.0f
    };

    // 2. generate a Vertex Buffer Object ID and bind it
    glGenBuffers(1, &VBOid);
    glBindBuffer(GL_ARRAY_BUFFER, VBOid);
    glBufferData(GL_ARRAY_BUFFER, sizeof(GLfloat) * 9, trianglevertices,
GL_STATIC_DRAW);

    // 3. activate and specify pointer to vertex array
    glEnableClientState(GL_VERTEX_ARRAY);

    // 4. draw the primitive object (0 = "use the currently bound buffer")
    glVertexPointer(3, GL_FLOAT, 0, 0);
    glDrawArrays(GL_TRIANGLES, 0, 3);

    // 5. disable client side capability for normal operation
    glDisableClientState(GL_VERTEX_ARRAY);

    // bind with 0, so switch back to normal pointer operation
    glBindBuffer(GL_ARRAY_BUFFER, 0);
    glDeleteBuffers(1, &VBOid);
}

void display() {
    glClear(GL_COLOR_BUFFER_BIT);
    glColor4f(.16f, .72f, .08f, 1.0f);
    triangle();
    glFlush();
}

Example 07 — glBufferData Usage Hints Compared
Concept: The same triangle shape drawn three times with GL_STATIC_DRAW, GL_DYNAMIC_DRAW, and
GL_STREAM_DRAW, to compare the usage-hint argument.

Expected output: Three triangles side by side, each built with a different glBufferData usage hint.

  #define GL_SILENCE_DEPRECATION
  #ifdef __APPLE__
  #include <GLUT/glut.h>
  #else
  #include <GL/glew.h>
  #include <GL/glut.h>
  #endif
  #include <iostream>
  #include <cmath>
  #include <cstdlib>
  using namespace std;

  /*
   * Example 07 - glBufferData Usage Hints
   * ------------------------------------------------------------
   * Concept: glBufferData's final argument is a USAGE HINT telling
   * OpenGL how the data will be used, so the driver can choose the best
   * memory location/strategy:
   *   GL_STATIC_DRAW - set once, drawn many times (most common for
   *                      unchanging geometry).
   *   GL_DYNAMIC_DRAW - changed repeatedly, drawn many times (used
   *                      later in this module for animated VBOs).
   *   GL_STREAM_DRAW - set once, drawn only a few times.
   * The RENDERED RESULT is identical either way - only the performance
   * characteristics differ. This example draws the same triangle three
   * times, each from a separately created VBO using a different hint,
   * purely to show the syntax side by side.
   */

  GLuint staticVBO, dynamicVBO, streamVBO;

  GLuint makeTriangleVBO(GLfloat offsetX, GLenum usage) {
      GLfloat verts[] = {
          offsetX + 0.0f, 0.75f, 0.0f,
          offsetX - 0.2f, 0.55f, 0.0f,



                                   Module 5 — Student Laboratory Manual | Page 12
        offsetX + 0.2f, 0.55f, 0.0f
    };
    GLuint id;
    glGenBuffers(1, &id);
    glBindBuffer(GL_ARRAY_BUFFER, id);
    glBufferData(GL_ARRAY_BUFFER, sizeof(GLfloat) * 9, verts, usage);
    return id;
}

void drawVBO(GLuint id) {
    glBindBuffer(GL_ARRAY_BUFFER, id);
    glEnableClientState(GL_VERTEX_ARRAY);
    glVertexPointer(3, GL_FLOAT, 0, 0);
    glDrawArrays(GL_TRIANGLES, 0, 3);
    glDisableClientState(GL_VERTEX_ARRAY);
    glBindBuffer(GL_ARRAY_BUFFER, 0);
}

void display() {
    glClear(GL_COLOR_BUFFER_BIT);

    staticVBO = makeTriangleVBO(-0.6f, GL_STATIC_DRAW);
    dynamicVBO = makeTriangleVBO(0.0f, GL_DYNAMIC_DRAW);
    streamVBO = makeTriangleVBO(0.6f, GL_STREAM_DRAW);

    glColor3f(0.9f, 0.3f, 0.3f);
    drawVBO(staticVBO);
    glColor3f(0.3f, 0.9f, 0.3f);
    drawVBO(dynamicVBO);
    glColor3f(0.3f, 0.3f, 0.9f);
    drawVBO(streamVBO);

    glDeleteBuffers(1, &staticVBO);
    glDeleteBuffers(1, &dynamicVBO);
    glDeleteBuffers(1, &streamVBO);

    glFlush();
}

int main(int argc, char** argv) {
    glutInit(&argc, argv);
    glutInitWindowSize(800, 600);

/*
 * Example 09 - A Second Buffer for Indices (GL_ELEMENT_ARRAY_BUFFER)
 * ------------------------------------------------------------
 * Concept: Just as GL_ARRAY_BUFFER holds vertex data on the GPU,
 * GL_ELEMENT_ARRAY_BUFFER holds INDEX data on the GPU. The pattern is
 * identical: glGenBuffers, glBindBuffer, glBufferData - just with a
 * different target and a GLubyte index array. Once bound,
 * glDrawElements's final argument becomes an OFFSET (0) instead of a
 * real pointer, exactly like glVertexPointer's did for GL_ARRAY_BUFFER.
 */

GLuint VBOid, VBOindex;

void triangle() {
    GLfloat trianglevertices[] = {
        0.0f, .75f, 0.0f,
        -0.75f, 0.0f, 0.0f,
        0.75f, 0.0f, 0.0f
    };
    GLubyte indices[] = { 0, 1, 2 };

    glGenBuffers(1, &VBOid);
    glBindBuffer(GL_ARRAY_BUFFER, VBOid);
    glBufferData(GL_ARRAY_BUFFER, sizeof(GLfloat) * 9, trianglevertices,
GL_STATIC_DRAW);

    glGenBuffers(1, &VBOindex);
    glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, VBOindex);
    glBufferData(GL_ELEMENT_ARRAY_BUFFER, sizeof(GLubyte) * 3, indices, GL_STATIC_DRAW);

    glEnableClientState(GL_VERTEX_ARRAY);
    glVertexPointer(3, GL_FLOAT, 0, 0);
    glDrawElements(GL_TRIANGLES, 3, GL_UNSIGNED_BYTE, 0);
    glDisableClientState(GL_VERTEX_ARRAY);

    glBindBuffer(GL_ARRAY_BUFFER, 0);
    glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
    glDeleteBuffers(1, &VBOid);
    glDeleteBuffers(1, &VBOindex);
}

Example 19 — Batching Many Shapes Into One VBO
Concept: 50 small triangles generated procedurally with sinf() and uploaded into ONE buffer, rendered with
a single glDrawArrays call.

Expected output: A ring or scattering of 50 small colored triangles, all drawn from one buffer in one draw

  #include <cmath>
  #include <cstdlib>
  using namespace std;

  /*
   * Example 19 - Batching Many Triangles Into ONE VBO
   * ------------------------------------------------------------
   * Concept: The real-world payoff of VBOs shows up at scale. Here, 50
   * small triangles are generated procedurally into ONE large array,
   * uploaded to the GPU in ONE glBufferData call, and rendered with ONE
   * glDrawArrays call - instead of 50 separate immediate-mode




                                   Module 5 — Student Laboratory Manual | Page 34
* glBegin/glEnd blocks or 50 separate small VBOs.
*/

const int TRIANGLE_COUNT = 50;
GLuint VBOid;

void batch() {
    GLfloat verts[TRIANGLE_COUNT * 9];
    for (int i = 0; i < TRIANGLE_COUNT; i++) {
        float t = (float)i / TRIANGLE_COUNT;
        float cx = -0.9f + t * 1.8f;
        float cy = 0.3f * sinf(t * 6.28f * 2.0f);
        int b = i * 9;
        verts[b + 0] = cx;        verts[b + 1] = cy + 0.03f; verts[b + 2] = 0.0f;
        verts[b + 3] = cx - 0.02f; verts[b + 4] = cy - 0.02f; verts[b + 5] = 0.0f;
        verts[b + 6] = cx + 0.02f; verts[b + 7] = cy - 0.02f; verts[b + 8] = 0.0f;
    }

    glGenBuffers(1, &VBOid);
    glBindBuffer(GL_ARRAY_BUFFER, VBOid);
    glBufferData(GL_ARRAY_BUFFER, sizeof(verts), verts, GL_STATIC_DRAW);

    glEnableClientState(GL_VERTEX_ARRAY);
    glVertexPointer(3, GL_FLOAT, 0, 0);
    glColor3f(0.2f, 0.8f, 0.9f);
    glDrawArrays(GL_TRIANGLES, 0, TRIANGLE_COUNT * 3);
    glDisableClientState(GL_VERTEX_ARRAY);

    glBindBuffer(GL_ARRAY_BUFFER, 0);
    glDeleteBuffers(1, &VBOid);
}

    glClear(GL_COLOR_BUFFER_BIT);
    batch();
    glFlush();
}

void animateStep(int value) {
    static float dy = -0.01f;

    glBindBuffer(GL_ARRAY_BUFFER, triPosVBO);
    GLfloat* mapped = (GLfloat*) glMapBuffer(GL_ARRAY_BUFFER, GL_READ_WRITE);
    if (mapped[1] > 0.75f || mapped[1] < 0.4f) dy = -dy;
    mapped[1] += dy; mapped[4] += dy; mapped[7] += dy;
    glUnmapBuffer(GL_ARRAY_BUFFER);
    glBindBuffer(GL_ARRAY_BUFFER, 0);

}
